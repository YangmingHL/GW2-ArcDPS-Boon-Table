function(download_dependency url name)
    set(hash_argument "")
    if(ARGC GREATER 2)
        set(hash_argument EXPECTED_HASH "SHA256=${ARGV2}")
    endif()
    file(DOWNLOAD "${url}" "${DEPS_DIR}/${name}"
        ${hash_argument} TLS_VERIFY ON STATUS download_status)
    list(GET download_status 0 download_code)
    if(NOT download_code EQUAL 0)
        message(FATAL_ERROR "Download failed for ${name}: ${download_status}")
    endif()
endfunction()
